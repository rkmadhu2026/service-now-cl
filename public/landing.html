Designed for developers

Configure via API-first workflows and declarative rules. Keep workflows versioned with your infrastructure. 

resource "relayroom_routing" "mute_minor" {
  tenant_id    = "demo-tenant-id"
  display_name = "Mute outbound on minor severity"
  rules = [{
    conditions = { severities = ["sev3"] }
    channels = {
      outbound_integrations_muted = true
    }
  }]
}
